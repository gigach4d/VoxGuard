module uart_audio_tx (
    input wire clk,
    input wire rst,
    input wire [15:0] audio_in,
    input wire audio_valid,
    output reg tx_pin
);
    parameter CLKS_PER_BIT = 12; // 1 Mbps at 12MHz
    
    reg [3:0] state = 0;
    reg [3:0] bit_idx = 0;
    reg [7:0] clk_cnt = 0;
    reg [15:0] tx_data = 0;

    always @(posedge clk) begin
        if (rst) begin
            state <= 0;
            tx_pin <= 1;
            clk_cnt <= 0;
        end else begin
            case (state)
                0: begin 
                    tx_pin <= 1;
                    if (audio_valid) begin
                        tx_data <= audio_in;
                        state <= 1; 
                    end
                end
                1: begin // Start HB
                    tx_pin <= 0;
                    if (clk_cnt == CLKS_PER_BIT-1) begin clk_cnt <= 0; state <= 2; bit_idx <= 0; end
                    else clk_cnt <= clk_cnt + 1;
                end
                2: begin // Data HB
                    tx_pin <= tx_data[15 - bit_idx];
                    if (clk_cnt == CLKS_PER_BIT-1) begin
                        clk_cnt <= 0;
                        if (bit_idx == 7) state <= 3;
                        else bit_idx <= bit_idx + 1;
                    end else clk_cnt <= clk_cnt + 1;
                end
                3: begin // Stop HB
                    tx_pin <= 1;
                    // THE FIX: Go to state 7 (Inter-byte delay) instead of state 4
                    if (clk_cnt == CLKS_PER_BIT-1) begin clk_cnt <= 0; state <= 7; end 
                    else clk_cnt <= clk_cnt + 1;
                end
                7: begin // INTER-BYTE DELAY (Allows RX to cleanly reset)
                    tx_pin <= 1;
                    if (clk_cnt == CLKS_PER_BIT-1) begin clk_cnt <= 0; state <= 4; end
                    else clk_cnt <= clk_cnt + 1;
                end
                4: begin // Start LB
                    tx_pin <= 0;
                    if (clk_cnt == CLKS_PER_BIT-1) begin clk_cnt <= 0; state <= 5; bit_idx <= 0; end
                    else clk_cnt <= clk_cnt + 1;
                end
                5: begin // Data LB
                    tx_pin <= tx_data[7 - bit_idx];
                    if (clk_cnt == CLKS_PER_BIT-1) begin
                        clk_cnt <= 0;
                        if (bit_idx == 7) state <= 6;
                        else bit_idx <= bit_idx + 1;
                    end else clk_cnt <= clk_cnt + 1;
                end
                6: begin // Stop LB
                    tx_pin <= 1;
                    if (clk_cnt == CLKS_PER_BIT-1) begin clk_cnt <= 0; state <= 0; end
                    else clk_cnt <= clk_cnt + 1;
                end
            endcase
        end
    end
endmodule
