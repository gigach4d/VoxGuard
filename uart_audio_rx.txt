module uart_audio_rx (
    input wire clk,
    input wire rst,
    input wire rx_pin,
    output reg [15:0] audio_out,
    output reg audio_valid
);
    parameter CLKS_PER_BIT = 12;
    parameter IDLE_TIMEOUT = 60; // 5us idle forces alignment reset
    
    reg [2:0] state = 0;
    reg [3:0] bit_idx = 0;
    reg [7:0] clk_cnt = 0;
    reg [7:0] idle_cnt = 0;
    reg [7:0] high_byte = 0;
    reg waiting_for_low = 0; 

    always @(posedge clk) begin
        if (rst) begin
            state <= 0;
            audio_valid <= 0;
            waiting_for_low <= 0;
        end else begin
            audio_valid <= 0;

            if (rx_pin == 1 && state == 0) begin
                if (idle_cnt < IDLE_TIMEOUT) idle_cnt <= idle_cnt + 1;
                else waiting_for_low <= 0; 
            end else begin
                idle_cnt <= 0;
            end

            case (state)
                0: begin 
                    if (rx_pin == 0) begin clk_cnt <= 0; state <= 1; end
                end
                1: begin 
                    if (clk_cnt == (CLKS_PER_BIT/2)-1) begin
                        if (rx_pin == 0) begin clk_cnt <= 0; state <= 2; bit_idx <= 0; end
                        else state <= 0; 
                    end else clk_cnt <= clk_cnt + 1;
                end
                2: begin 
                    if (clk_cnt == CLKS_PER_BIT-1) begin
                        clk_cnt <= 0;
                        if (!waiting_for_low) high_byte[7 - bit_idx] <= rx_pin;
                        else audio_out[7 - bit_idx] <= rx_pin; 
                        
                        if (bit_idx == 7) state <= 3;
                        else bit_idx <= bit_idx + 1;
                    end else clk_cnt <= clk_cnt + 1;
                end
                3: begin 
                    if (clk_cnt == CLKS_PER_BIT-1) begin
                        clk_cnt <= 0;
                        state <= 0;
                        if (!waiting_for_low) begin
                            waiting_for_low <= 1; 
                        end else begin
                            audio_out[15:8] <= high_byte; 
                            audio_valid <= 1;
                            waiting_for_low <= 0;
                        end
                    end else clk_cnt <= clk_cnt + 1;
                end
            endcase
        end
    end
endmodule
